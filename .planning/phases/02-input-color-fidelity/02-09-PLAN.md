async def _modifier_periodic_safety_loop(self):
    import time
    while not self._stopped:
        await asyncio.sleep(2)
        now = time.monotonic()
        if _should_fire_periodic_reset(now, self._last_key_event_at or now, self._last_key_event_had_modifiers):
            self._input_injector.reset_modifiers()
            log.info("input.periodic_reset_modifiers", quiet_seconds=(now - self._last_key_event_at))
            self._last_key_event_at = now

def _should_fire_periodic_reset(now: float, last_event: float, last_had_modifiers: bool) -> bool:
    if last_had_modifiers:
        return False
    return (now - last_event) >= 10.0
