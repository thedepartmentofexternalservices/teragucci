@dataclass
class KeyResetModifiersMsg:

    Fired on 4 triggers (D-11): focusOutEvent, reconnect, server-periodic-safety,
    F9 panic. Server dispatches to InputInjector.reset_modifiers() idempotently.
    """

    reason: str = "unknown"  # "focus_out" | "reconnect" | "periodic" | "panic_f9" | "unknown"

    def to_json(self) -> str:
        return json.dumps(asdict(self))


@dataclass
class TextCommitMsg:
    """Client → Server: IME / dead-key commit string (D-15).

    xdotool type (Linux) or CGEventKeyboardSetUnicodeString (macOS).
    """
    type: str = MsgType.TEXT_COMMIT
    text: str = ""

    def to_json(self) -> str:
        return json.dumps(asdict(self))
