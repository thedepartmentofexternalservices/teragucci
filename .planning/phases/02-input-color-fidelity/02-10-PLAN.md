"""Phase 2 INPUT-08: IOHIDUserDevice-backed virtual Wacom tablet for macOS server.

Replaces the mouse-click fallback in mac_input_injector.pen_event (D-09).
Public contract matches pen_event() in mac_input_injector.py; platform_backends
composes MacInputInjector + MacPenInjector into a single InputInjector facade.
"""
from __future__ import annotations

import atexit
import signal
import struct
import threading
from typing import Optional

from common.logging import get_logger

log = get_logger("server.mac_pen_injector")

try:
    import objc
    from Foundation import NSData, NSMutableDictionary
    IOKit = objc.loadBundle(
        "IOKit", globals(),
        bundle_path="/System/Library/Frameworks/IOKit.framework",
    )
    _HAS_IOKIT = True
    _import_error = None
except Exception as _e:
    _HAS_IOKIT = False
    _import_error = _e


class MacPenInjectorError(RuntimeError):
    """Raised when IOKit / PyObjC is unavailable or device creation fails."""


PEN_MAX_PRESSURE = 8191   # Wacom 8192-level pressure
PEN_MAX_TILT = 127        # signed 7-bit per axis
PEN_COORD_MAX = 65535     # 16-bit absolute


# Linuxwacom-derived Intuos Pro HID report descriptor (verified pattern from
# linuxwacom/wacom-hid-descriptors Intuos Pro profile; pressure is 2048 levels / 11-bit,
# extended to 13-bit / 8192 levels where supported — we advertise 13-bit / 8191 here).
# D-07/D-08 spike may produce a slightly different descriptor (Cintiq Pro 24 is very
# similar — minor pressure-range differences). If the spike chooses a different byte
# sequence, Task 0 writes those bytes to server/wacom_descriptors.py::INTUOS_PRO_DESCRIPTOR
# and Task 1 imports from there; otherwise default to the descriptor below.
WACOM_REPORT_DESCRIPTOR = bytes([
    0x05, 0x0D,              # Usage Page (Digitizers)
    0x09, 0x02,              # Usage (Pen)
    0xA1, 0x01,              # Collection (Application)
    0x85, 0x10,              #   Report ID (16)
    0x09, 0x20,              #   Usage (Stylus)
    0xA1, 0x00,              #   Collection (Physical)
    0x09, 0x42,              #     Usage (Tip Switch)
    0x09, 0x44,              #     Usage (Barrel Switch)
    0x09, 0x45,              #     Usage (Eraser)
    0x09, 0x3C,              #     Usage (Invert)
    0x09, 0x32,              #     Usage (In Range)
    0x15, 0x00,              #     Logical Minimum (0)
    0x25, 0x01,              #     Logical Maximum (1)
    0x75, 0x01,              #     Report Size (1)
    0x95, 0x05,              #     Report Count (5)
    0x81, 0x02,              #     Input (Data,Var,Abs)
    0x95, 0x03,              #     Report Count (3)  — padding
    0x81, 0x03,              #     Input (Cnst,Var,Abs)
    0x05, 0x01,              #     Usage Page (Generic Desktop)
    0x09, 0x30,              #     Usage (X)
    0x09, 0x31,              #     Usage (Y)
    0x16, 0x00, 0x00,        #     Logical Minimum (0)
    0x26, 0xFF, 0x7F,        #     Logical Maximum (32767)
    0x75, 0x10,              #     Report Size (16)
    0x95, 0x02,              #     Report Count (2)
    0x81, 0x02,              #     Input (Data,Var,Abs)
    0x05, 0x0D,              #     Usage Page (Digitizers)
    0x09, 0x30,              #     Usage (Tip Pressure)
    0x26, 0xFF, 0x1F,        #     Logical Maximum (8191)  — 13-bit pressure
    0x75, 0x10,              #     Report Size (16)
    0x95, 0x01,              #     Report Count (1)
    0x81, 0x02,              #     Input (Data,Var,Abs)
    0x09, 0x3D,              #     Usage (X Tilt)
    0x09, 0x3E,              #     Usage (Y Tilt)
    0x15, 0xC0,              #     Logical Minimum (-64)
    0x25, 0x3F,              #     Logical Maximum (63)
    0x75, 0x08,              #     Report Size (8)
    0x95, 0x02,              #     Report Count (2)
    0x81, 0x02,              #     Input (Data,Var,Abs)
    0xC0,                    #   End Collection
    0xC0,                    # End Collection
])


# D-07/D-08 spike fallback: if the spike produces a different descriptor (Cintiq
# Pro 24 deviates minimally from Intuos Pro; pressure range may differ), Task 0
# writes the chosen bytes to server/wacom_descriptors.py::INTUOS_PRO_DESCRIPTOR
# and Task 1 imports them in preference to the above default:
try:
    from server.wacom_descriptors import INTUOS_PRO_DESCRIPTOR as _SPIKE_DESCRIPTOR
    WACOM_REPORT_DESCRIPTOR = _SPIKE_DESCRIPTOR  # noqa: F811 — spike-authored override
except ImportError:
    pass  # keep the default linuxwacom-derived descriptor above


class MacPenInjector:
    def __init__(self, vendor_id: int = 0x056A, product_id: int = 0x034D):
        if not _HAS_IOKIT:
            raise MacPenInjectorError(
                f"IOKit / PyObjC not available: {_import_error}. "
                "pip install pyobjc-framework-IOKit"
            )
        self._vendor_id = vendor_id
        self._product_id = product_id
        self._device = None
        self._lock = threading.Lock()
        self._stopped = False
        self._atexit_registered = False

    def start(self) -> None:
        props = NSMutableDictionary.alloc().init()
        props["VendorID"] = self._vendor_id
        props["ProductID"] = self._product_id
        props["ReportDescriptor"] = NSData.dataWithBytes_length_(
            WACOM_REPORT_DESCRIPTOR, len(WACOM_REPORT_DESCRIPTOR)
        )
        props["PrimaryUsagePage"] = 0x0D
        props["PrimaryUsage"] = 0x02
        self._device = IOKit.IOHIDUserDeviceCreate(None, props)
        if self._device is None:
            raise MacPenInjectorError("IOHIDUserDeviceCreate returned NULL")
        if not self._atexit_registered:
            atexit.register(self._cleanup_on_exit)
            try:
                signal.signal(signal.SIGTERM, self._signal_cleanup)
                signal.signal(signal.SIGINT,  self._signal_cleanup)
            except (ValueError, OSError):
                pass  # Not the main thread -- atexit alone covers us
            self._atexit_registered = True
        log.info("mac_pen_injector.started", vendor_id=hex(self._vendor_id),
                 product_id=hex(self._product_id))

    def pen_event(self, x: float, y: float, pressure: float,
                  tilt_x: float = 0.0, tilt_y: float = 0.0,
                  rotation: float = 0.0, button: int = 0, pressed: bool = False,
                  hovering: bool = False, pen_type: str = "pen") -> None:
        """IOHIDUserDevice-backed pen injection (D-09)."""
        if self._stopped or self._device is None:
            return
        # Pack HID report bytes per the descriptor. Field order & widths must
        # match WACOM_REPORT_DESCRIPTOR exactly; the D-07 spike confirms.
        xi = max(0, min(PEN_COORD_MAX, int(x * PEN_COORD_MAX)))
        yi = max(0, min(PEN_COORD_MAX, int(y * PEN_COORD_MAX)))
        pi = max(0, min(PEN_MAX_PRESSURE, int(pressure * PEN_MAX_PRESSURE)))
        txi = max(-PEN_MAX_TILT, min(PEN_MAX_TILT, int(tilt_x * PEN_MAX_TILT)))
        tyi = max(-PEN_MAX_TILT, min(PEN_MAX_TILT, int(tilt_y * PEN_MAX_TILT)))
        # Report layout (matches WACOM_REPORT_DESCRIPTOR exactly):
        #   offset 0:  Report ID (0x10)
        #   offset 1:  buttons/flags byte — Tip | Barrel | Eraser | Invert | InRange in low 5 bits
        #   offset 2:  X (u16 LE, 0..32767)
        #   offset 4:  Y (u16 LE, 0..32767)
        #   offset 6:  Pressure (u16 LE, 0..8191 — 13-bit, zero-extended)
        #   offset 8:  Tilt X (i8, -64..63)
        #   offset 9:  Tilt Y (i8, -64..63)
        # Total = 10 bytes per report.
        flags = 0
        if pressed: flags |= 0x01        # Tip Switch
        if button == 1: flags |= 0x02    # Barrel Switch
        if pen_type == "eraser": flags |= 0x04  # Eraser
        if pen_type == "eraser": flags |= 0x08  # Invert
        if hovering or not pressed: flags |= 0x10  # In Range

        # Re-clamp pressure to the 13-bit range the descriptor advertises.
        pressure_u16 = max(0, min(8191, int(pressure * 8191)))
        # X and Y are 16-bit abs 0..32767 per descriptor Logical Maximum 0x7FFF.
        x_u16 = max(0, min(0x7FFF, int(x * 0x7FFF)))
        y_u16 = max(0, min(0x7FFF, int(y * 0x7FFF)))
        # Tilt range -64..63 (i8); clamp to the descriptor's logical bounds.
        tilt_x_i8 = max(-64, min(63, int(tilt_x * 64)))
        tilt_y_i8 = max(-64, min(63, int(tilt_y * 64)))

        report = struct.pack(
            "<BBHHHbb",
            0x10,                           # Report ID
            flags,                          # Tip | Barrel | Eraser | Invert | InRange in low 5 bits
            x_u16, y_u16,                   # 16-bit abs positions
            pressure_u16,                   # 13-bit pressure (0..8191) zero-extended to 16-bit
            tilt_x_i8, tilt_y_i8,           # -64..63
        )
        IOKit.IOHIDUserDeviceHandleReport(self._device, report, len(report))

    def stop(self) -> None:
        if self._stopped:
            return
        self._stopped = True
        if self._device is not None:
            try:
                IOKit.IOHIDUserDeviceInvalidate(self._device)
            finally:
                self._device = None
        log.info("mac_pen_injector.stopped")

    def _cleanup_on_exit(self):
        self.stop()

    def _signal_cleanup(self, signum, frame):
        self.stop()
```

# Phase 2 D-19: separate PenFSM (orthogonal to ClientFSM/ServerFSM).
class PenFSM(StateMachine):
    out_of_proximity = State("out_of_proximity", initial=True)
    in_proximity = State("in_proximity")

    enter_proximity = (
        out_of_proximity.to(in_proximity)
        | in_proximity.to(in_proximity)  # idempotent per D-19
    )
    leave_proximity = (
        in_proximity.to(out_of_proximity)
        | out_of_proximity.to(out_of_proximity)
    )
```
