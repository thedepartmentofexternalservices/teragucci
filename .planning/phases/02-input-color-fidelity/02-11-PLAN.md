```
"""D-20: read-only TCC DB inspection + Wacom driver detection.

Never writes, never bypasses. sqlite3 mode=ro over the file URI scheme.
"""
from __future__ import annotations

import pathlib
import sqlite3
import subprocess
from typing import Any

from common.logging import get_logger

log = get_logger("client.tcc_detect")

TCC_DB_PATH = pathlib.Path("~/Library/Application Support/com.apple.TCC/TCC.db").expanduser()

SERVICE_INPUT_MONITORING = "kTCCServiceListenEvent"
SERVICE_ACCESSIBILITY = "kTCCServiceAccessibility"


def read_tcc_status(
    client_bundle_id: str = "com.teraguchi.client",
    wacom_pattern: str = "com.wacom.",
) -> dict[str, Any]:
    result = {
        "input_monitoring_granted": False,
        "accessibility_granted_for_wacom": False,
        "wacom_driver_installed": False,
        "raw_rows": [],
        "tcc_db_readable": False,
    }
    if not TCC_DB_PATH.exists():
        log.info("tcc_detect.db_not_found", path=str(TCC_DB_PATH))
        return result
    try:
        uri = f"file:{TCC_DB_PATH}?mode=ro"
        conn = sqlite3.connect(uri, uri=True, timeout=1.0)
        result["tcc_db_readable"] = True
    except sqlite3.Error as e:
        log.warning("tcc_detect.open_failed", error=str(e))
        return result
    try:
        cursor = conn.execute(
            "SELECT service, client, auth_value FROM access "
            "WHERE service IN (?, ?)",
            (SERVICE_INPUT_MONITORING, SERVICE_ACCESSIBILITY),
        )
        for service, client, auth_value in cursor:
            result["raw_rows"].append((service, client, auth_value))
            if service == SERVICE_INPUT_MONITORING and client == client_bundle_id:
                result["input_monitoring_granted"] = (auth_value >= 2)
            if service == SERVICE_ACCESSIBILITY and wacom_pattern in client:
                result["accessibility_granted_for_wacom"] = (auth_value >= 2)
    finally:
        conn.close()
    result["wacom_driver_installed"] = _detect_wacom_driver()
    return result


def _detect_wacom_driver() -> bool:
    """Lightweight check: look for /Applications/Wacom Tablet.app or /Library/Application Support/Tablet."""
    for p in ("/Applications/Wacom Tablet.app",
              "/Library/Application Support/Tablet"):
        if pathlib.Path(p).exists():
            return True
    return False


def list_connected_wacom_devices() -> list[str]:
    """system_profiler SPUSBDataType | grep Wacom. Returns list of product strings."""
    try:
        out = subprocess.run(
            ["system_profiler", "SPUSBDataType", "-xml"],
            capture_output=True, text=True, timeout=5, check=False,
        )
        matches = [line.strip() for line in out.stdout.splitlines() if "Wacom" in line]
        return matches[:20]
    except Exception:
        return []


SYSTEM_SETTINGS_INPUT_MONITORING = "x-apple.systempreferences:com.apple.preference.security?Privacy_ListenEvent"
SYSTEM_SETTINGS_ACCESSIBILITY  = "x-apple.systempreferences:com.apple.preference.security?Privacy_Accessibility"
WACOM_DRIVER_URL = "https://www.wacom.com/en-us/support/product-support/drivers"
```
