Step B -- Change the decode hot path. Find the current `decode_frame` / `decode_packet` function. Today it likely does `frame.to_ndarray(format='rgb24')` somewhere. Replace with direct plane extraction for yuv420p10le/p010le frames:

```
def decode_frame(self, packet_bytes: bytes):
    """Decode next HEVC packet -> (y_bytes, uv_bytes, width, height).

    Phase 2: returns 10-bit planes directly to VideoBlitWidget. Phase 1
    rgb24 coercion is GONE on the video hot path (see PITFALLS.md cp.5
    Qt QImage 8-bit coercion).
    """
    packet = av.Packet(packet_bytes)
    for frame in self._decoder.decode(packet):
        fmt = frame.format.name
        # D-01 cp.5: first-frame format assertion under Main10
        if self._negotiated_main10 and fmt not in ("p010le", "yuv420p10le"):
            log.error("video_decoder.p010_format_assertion_failed",
                      expected=["p010le", "yuv420p10le"], got=fmt)
            raise RuntimeError(
                f"Main10 negotiated but decoder returned {fmt!r}; "
                "check FFmpeg/PyAV build against system FFmpeg 7.1+")
        y_bytes, uv_bytes = self._extract_planes_p010(frame)
        return y_bytes, uv_bytes, frame.width, frame.height
    return None


def _extract_planes_p010(self, frame):
    """Extract Y + UV planes as raw bytes from a p010le/yuv420p10le AVFrame.
    For p010le (semi-planar Y then interleaved UV): planes[0] is Y, planes[1] is UV.
    For yuv420p10le (planar Y, U, V): interleave U+V into the UV buffer."""
    if frame.format.name == "p010le":
        y = bytes(frame.planes[0])
        uv = bytes(frame.planes[1])
        return y, uv
    # yuv420p10le -- interleave U + V
    y = bytes(frame.planes[0])
    u = bytes(frame.planes[1])
    v = bytes(frame.planes[2])
    uv = bytearray(len(u) + len(v))
    uv[0::2] = u
    uv[1::2] = v
    return y, bytes(uv)
```
