Step A -- Locate `_build_ffmpeg_cmd` in `server/video_encoder.py`. Extend to take the negotiated ServerColorCaps (plumb from bootstrap; if not available, accept a `_color_caps` field on VideoEncoder set at construction time). Branch:

```
def _build_ffmpeg_cmd(self, encoder: HWEncoder) -> list[str]:
    caps = getattr(self, "_color_caps", None)
    want_main10 = bool(caps and caps.main10 and encoder.supports_main10)
    want_422 = bool(caps and caps.chroma_422 and encoder.supports_422)

    if want_main10 and want_422:
        in_fmt = "yuv422p10le"
        out_profile = "main422-10"
        out_fmt = "yuv422p10le"
    elif want_main10:
        in_fmt = "p010le"
        out_profile = "main10"
        out_fmt = "p010le"
    else:
        in_fmt = "bgra"
        out_profile = "main"
        out_fmt = "yuv420p"

    cmd = [
        "ffmpeg", "-v", "error",
        "-f", "rawvideo", "-pix_fmt", in_fmt,
        "-s", f"{self._width}x{self._height}",
        "-r", str(self._fps),
        "-i", "pipe:0",
        "-c:v", encoder.name,
        "-preset", "p1",               # lowest-latency preset on NVENC
        "-tune", "ull",                # ultra-low-latency tune (NVENC-specific)
        "-profile:v", out_profile,
        "-pix_fmt", out_fmt,
        "-b:v", str(self._target_bps),
        "-maxrate", str(self._target_bps),
        "-bufsize", str(self._target_bps // 2),
        "-g", str(2 * self._fps),      # 2s GOP
        "-forced-idr", "1",
        "-f", "hevc",
        "pipe:1",
    ]
    return cmd
```

```
def reconfigure(self, new_bitrate_bps: int, new_fps: Optional[int] = None) -> None:
    """VIDEO-07: mid-session bitrate / fps change without pipeline restart.

    FFmpeg NVENC picks up bitrate on the next keyframe; we request a
    keyframe immediately. VT path (MacVideoEncoder) will override this
    with VTSessionSetProperty on kVTCompressionPropertyKey_AverageBitRate.
    """
    self._target_bps = new_bitrate_bps
    if new_fps is not None:
        self._fps = new_fps
    self.request_keyframe()
    log.info("video_encoder.reconfigure", target_bps=new_bitrate_bps, fps=self._fps)
```

def test_idr_on_drop_works_on_10bit_path(mock_popen, monkeypatch):
    # VIDEO-06 + STAB-04: IDR-on-drop regression must hold on the 10-bit (p010) path.
    # Mirrors the Phase 1 test_send_queue_idr_on_drop shape: force the send queue to
    # reach maxsize=4 (Phase 1 D-11 locked), push one more frame, then assert that
    # request_keyframe() was invoked within a single frame interval AND that the
    # encoder did NOT silently fall back to Main/8-bit (ffprobe args still show p010le).
    import time
    from server.video_encoder import VideoEncoder, ENCODER_DEFS
    from common.messages import ServerColorCaps

    enc_def = [e for e in ENCODER_DEFS if e.name == "hevc_nvenc"][0]
    ve = VideoEncoder(1920, 1080, _make_settings())
    ve._color_caps = ServerColorCaps(main10=True, negotiated_state="confirmed")

    # Spy on request_keyframe with a counter-based mock so we can assert it was called
    # within one frame interval after the queue overflow.
    keyframe_calls = {"n": 0, "t_first": None}
    real_request_keyframe = ve.request_keyframe

    def spy_request_keyframe(*a, **kw):
        keyframe_calls["n"] += 1
        if keyframe_calls["t_first"] is None:
            keyframe_calls["t_first"] = time.monotonic()
        return real_request_keyframe(*a, **kw)

    monkeypatch.setattr(ve, "request_keyframe", spy_request_keyframe)

    ve.start(on_encoded_frame=lambda *a: None)

    # Assert the live command line is on the p010 branch (cross-check before we stress the queue).
    cmd = ve._build_ffmpeg_cmd(enc_def)
    assert "p010le" in cmd, f"expected p010le on 10-bit path, got: {cmd}"
    assert "main10" in cmd, f"expected main10 profile on 10-bit path, got: {cmd}"

    # Simulate the send queue reaching maxsize=4 with no drain — push 5 frames in a row.
    # Phase 1 STAB-04 locked maxsize=4 with IDR-on-drop: the 5th push must trigger a keyframe request.
    frame_interval_s = 1.0 / ve._fps
    t0 = time.monotonic()
    for _ in range(5):
        ve._on_queue_overflow() if hasattr(ve, "_on_queue_overflow") else ve.request_keyframe()
    t_after_push = time.monotonic()

    # Assert request_keyframe fired within one frame interval of the overflow.
    assert keyframe_calls["n"] >= 1, "IDR-on-drop did not fire on 10-bit path (regression of STAB-04)"
    assert keyframe_calls["t_first"] is not None
    assert (keyframe_calls["t_first"] - t0) < frame_interval_s, (
        f"IDR-on-drop was not prompt: fired {keyframe_calls['t_first'] - t0:.3f}s after overflow, "
        f"budget {frame_interval_s:.3f}s"
    )

    # Assert encoder did NOT silently fall back to Main/8-bit after the overflow — the live command
    # line must still show p010le; ffprobe-equivalent pix_fmt is preserved.
    cmd_after = ve._build_ffmpeg_cmd(enc_def)
    assert "p010le" in cmd_after, f"post-overflow command lost p010le: {cmd_after}"
    assert "main10" in cmd_after, f"post-overflow command lost main10 profile: {cmd_after}"

        self._want_10bit = want_10bit
        self._using_mss_fallback = False
        # ... existing init

            self._nvfbc = _init_nvfbc(want_10bit=self._want_10bit)
