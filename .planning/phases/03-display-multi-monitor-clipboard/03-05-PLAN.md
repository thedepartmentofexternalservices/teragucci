    ```python
    async def run(self) -> None:
        while self._running:
            await asyncio.sleep(1.0)
            runtime = self._runtime
            if not runtime.capture:
                continue
            pending = getattr(runtime.capture, "_hotplug_pending", False)
            if not (pending or runtime.capture.detect_hotplug()):
                continue
            if pending:
                runtime.capture._hotplug_pending = False

            monitors_data = [asdict(m) for m in runtime.capture.list_monitors()]

            # D-09 — re-apply per-session capture_mode and harvest degradations
            # BEFORE the broadcast so the MonitorListMsg carries the fallback
            # events with the new geometry.
            degradations = []
            for ws, cs in list(runtime.clients.items()):
                if not cs.authenticated:
                    continue
                if getattr(cs, "capture_mode", "mirror_all") == "mirror_all":
                    continue
                previous_pick = getattr(cs, "picked_monitor_name", "")
                ok = runtime.apply_capture_mode(
                    cs, cs.capture_mode, cs.picked_monitor_id, cs.picked_monitor_name,
                )
                if not ok and previous_pick:
                    # Fallback fired — emit a degradation event for this client.
                    degradations.append({
                        "client_token": getattr(cs, "client_token", "") or getattr(cs, "session_id", ""),
                        "previous_pick": previous_pick,
                        "now_showing": getattr(cs, "picked_monitor_name", "primary"),
                    })

            msg_json = MonitorListMsg(
                monitors=monitors_data,
                degradations=degradations,
            ).to_json()

            for ws, cs in list(runtime.clients.items()):
                if cs.authenticated:
                    try:
                        await cs.enqueue(msg_json)
                    except Exception:
                        pass

            if runtime.encoder:
                runtime.encoder_lifecycle.restart()

            logger.info(
                "monitor_hotplug.broadcast monitor_count=%d degradation_count=%d",
                len(monitors_data), len(degradations),
            )
            if degradations:
                logger.info("monitor_hotplug.fallback events=%d", len(degradations))
    ```
