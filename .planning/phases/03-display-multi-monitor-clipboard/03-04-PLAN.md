    ```python
    def _widget_to_remote(self, x: float, y: float) -> tuple:
        """Phase 3 D-05 — convert widget coords to BOTH normalized floats
        (Phase 1/2 wire compat) AND server physical-pixel integers.

        Returns (rx_norm, ry_norm, server_x_px, server_y_px). Server
        prefers integer fields when present (server_x >= 0); normalized
        floats remain for backwards-compat with pre-Phase-3 servers."""
        dpr = self._current_dpr  # cached; refreshed on screenChanged
        if not self._monitor_regions:
            # Simple: widget → full remote desktop
            rx = (x - self._offset_x) / (self._scale_x * self._remote_width)
            ry = (y - self._offset_y) / (self._scale_y * self._remote_height)
            rx = max(0.0, min(1.0, rx))
            ry = max(0.0, min(1.0, ry))
            sx = int(round(rx * self._remote_width))
            sy = int(round(ry * self._remote_height))
            return rx, ry, sx, sy

        # Composite mode: walk monitor regions
        cx = (x - self._offset_x) / self._scale_x
        cy = (y - self._offset_y) / self._scale_y
        composite_x = 0
        for region in self._monitor_regions:
            rw = region["width"]
            rh = region["height"]
            if cx < composite_x + rw:
                local_x = cx - composite_x
                local_y = cy
                desktop_x = region["x"] + local_x
                desktop_y = region["y"] + local_y
                rx = max(0.0, min(1.0, desktop_x / self._remote_width))
                ry = max(0.0, min(1.0, desktop_y / self._remote_height))
                sx = int(round(desktop_x))
                sy = int(round(desktop_y))
                return rx, ry, sx, sy
            composite_x += rw

        # Past the last monitor — clamp to last region's right edge
        last = self._monitor_regions[-1]
        sx = last["x"] + last["width"] - 1
        sy = int(round(cy))
        rx = max(0.0, min(1.0, sx / self._remote_width))
        ry = max(0.0, min(1.0, sy / self._remote_height if self._remote_height else 0))
        return rx, ry, sx, sy
    ```
    NOTE the DPR multiplication is implicit: `self._remote_width` and `self._remote_height` are server physical pixels (from MonitorListMsg), and `desktop_x` lives in that space already. The DPR matters when computing `cx`/`cy` from widget coords because `self._scale_x` (set by `_update_scaling`) was computed using DPR. The cleaner path: cache `self._current_dpr` from `_current_screen_dpr()` and feed it into `_update_scaling`.

    **Step B — Add per-screen DPR helpers and screenChanged hook.** Per RESEARCH Example 1 L546-576:
    ```python
    def _current_screen_dpr(self) -> float:
        """D-06 — return DPR of the QScreen the viewer widget is currently on.

        Always look up the CURRENT screen's DPR — never cache the primary's.
        Re-evaluated on QWindow.screenChanged. Falls back to devicePixelRatioF()
        when windowHandle() is None (pre-show edge case)."""
        win = self.window().windowHandle() if self.window() else None
        if win is None:
            return float(self.devicePixelRatioF())
        screen = win.screen()
        if screen is None:
            return float(self.devicePixelRatioF())
        return float(screen.devicePixelRatio())

    def _connect_screen_changed(self):
        """Wire screenChanged signal AFTER the widget is shown.

        windowHandle() is None until Qt creates the native window — so
        connect inside showEvent. Idempotent: safe to call multiple times."""
        win = self.window().windowHandle() if self.window() else None
        if win is None:
            return
        try:
            # Disconnect any prior connection (in case showEvent fires twice)
            win.screenChanged.disconnect(self._on_screen_changed)
        except (TypeError, RuntimeError):
            pass
        win.screenChanged.connect(self._on_screen_changed)

    def _on_screen_changed(self, screen):
        """D-06 — recompute scaling cache when widget migrates between screens.

        Symmetric to Phase 2 D-19 focusIn pen-proximity re-synth (Pitfall 2):
        re-emit pen_proximity if pen was in range so server PenFSM doesn't
        lose state across the screen change."""
        old_dpr = getattr(self, "_current_dpr", 1.0)
        self._current_dpr = self._current_screen_dpr()
        self._update_scaling()
        if old_dpr != self._current_dpr:
            logger.info(
                "viewer.screen_changed dpr=%s screen=%s",
                self._current_dpr,
                screen.name() if screen else "unknown",
            )
        # Pitfall 2 — re-synth pen proximity on screen migration via the
        # canonical Phase 2 helper at client/viewer.py::990 _emit_pen_proximity
        # (centralized dict construction so all callers — focusIn re-synth,
        # showEvent, screenChanged — emit the same shape).
        if getattr(self, "_pen_was_in_proximity", False):
            self._emit_pen_proximity(
                in_proximity=True,
                pen_type=getattr(self, "_last_pen_type", "pen"),
            )
        # F12 dev overlay — refresh if visible
        if getattr(self, "_coord_overlay", None) and self._coord_overlay.isVisible():
            self._coord_overlay.update_screen(self._current_dpr,
                                              screen.name() if screen else "unknown")
    ```
