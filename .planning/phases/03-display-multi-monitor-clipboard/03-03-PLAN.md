    ```python
    from typing import Optional, Tuple
    import numpy as np

    def capture_raw_bgra_with_crop(
        self, crop: Optional[Tuple[int, int, int, int]] = None
    ) -> bytes:
        """D-02 — full-virtual-desktop capture + optional GPU-side crop.

        crop is (x, y, w, h) in server physical pixels, or None to pass through.
        Validates the crop fits inside the captured frame; clamps if not.
        Returns the cropped BGRA bytes; degenerate crops return a single
        black pixel rather than crashing the encoder feed.
        """
        raw = self.capture_raw_bgra()  # unchanged Phase 2 path
        if crop is None:
            return raw
        x, y, w, h = crop
        x = max(0, min(x, self.width))
        y = max(0, min(y, self.height))
        w = max(0, min(w, self.width - x))
        h = max(0, min(h, self.height - y))
        if w == 0 or h == 0:
            return b"\x00\x00\x00\xff"
        arr = np.frombuffer(raw, dtype=np.uint8).reshape(self.height, self.width, 4)
        return arr[y:y+h, x:x+w].tobytes()
    ```

       ```python
       def apply_capture_mode(
           self,
           session,                    # ClientSession-like object
           mode: str,
           picked_id: int,
           picked_name: str,
       ) -> bool:
           """Phase 3 D-02 — set the per-session capture crop based on mode.

           Returns True if the request was honored as-asked, False if it
           was degraded (e.g. picked monitor missing → fell back to primary).
           Logs structured telemetry for D-18 tracking.
           """
           if mode not in ("single", "mirror_all", "pick_one"):
               logger.warning("session.invalid_capture_mode mode=%r → mirror_all", mode)
               mode = "mirror_all"
           session.capture_mode = mode
           session.capture_mode_degraded = False
           if mode == "mirror_all":
               session.crop_rect = None  # full-virtual-desktop pass-through
               session.picked_monitor_id = -1
               session.picked_monitor_name = ""
               return True
           if not self.capture:
               session.crop_rect = None
               return True  # capture not initialized yet; defer
           monitors = self.capture.list_monitors()
           # Skip the virtual-desktop entry (id=0 by convention; per
           # screen_capture.py list_monitors lines 201-204 the first entry
           # is "All Monitors (Virtual Desktop)")
           non_virtual = [m for m in monitors if m.id != 0]
           target = None
           # Prefer id match (D-04 belt+suspenders: id wins, name fallback)
           if picked_id >= 0:
               for m in non_virtual:
                   if m.id == picked_id:
                       target = m
                       break
           if target is None and picked_name:
               for m in non_virtual:
                   if m.name == picked_name:
                       target = m
                       break
           if target is None:
               # Fall back to primary (D-09 foundation; Plan 05 wires the toast)
               primary = next((m for m in non_virtual if m.primary), non_virtual[0] if non_virtual else None)
               if primary is None:
                   logger.warning("session.no_monitors_for_capture_mode mode=%s", mode)
                   session.crop_rect = None
                   return False
               target = primary
               session.capture_mode_degraded = True
               logger.warning(
                   "session.capture_mode_degraded requested_id=%d requested_name=%r → primary=%s",
                   picked_id, picked_name, target.name,
               )
           session.crop_rect = (target.x, target.y, target.width, target.height)
           session.picked_monitor_id = target.id
           session.picked_monitor_name = target.name
           # If single mode, even with no picked id we want to crop to primary
           # (D-01 — "single monitor" means show one, defaulting to primary).
           if mode == "single" and not session.capture_mode_degraded:
               primary = next((m for m in non_virtual if m.primary), non_virtual[0] if non_virtual else None)
               if primary:
                   session.crop_rect = (primary.x, primary.y, primary.width, primary.height)
                   session.picked_monitor_id = primary.id
                   session.picked_monitor_name = primary.name
           logger.info(
               "session.capture_mode_applied mode=%s crop_rect=%s degraded=%s",
               mode, session.crop_rect, session.capture_mode_degraded,
           )
           return not session.capture_mode_degraded
       ```
