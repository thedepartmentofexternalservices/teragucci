    **Step C — Add ModeSelector sub-widget to client/main_window.py::ConnectionDialog.** Insert AFTER the existing `swap_cmd_ctrl_check` row (lines 116-124), BEFORE the OK/Cancel buttons. Per UI-SPEC Surface 1 + 4, copy verbatim:

    ```python
    # Phase 3 D-01 / D-04 — Monitor mode picker (per-session, locked at connect)
    from PySide6.QtCore import Signal as _Signal
    from PySide6.QtWidgets import QGroupBox, QRadioButton, QButtonGroup, QVBoxLayout, QLabel
    from client.monitor_selector import MonitorSelector

    class ModeSelector(QWidget):
        # signal carries (mode, picked_monitor_id, picked_monitor_name)
        mode_changed = _Signal(str, int, str)

        def __init__(self, parent=None):
            super().__init__(parent)
            v = QVBoxLayout(self)
            v.setContentsMargins(0, 0, 0, 0)

            heading = QLabel("Monitor mode")
            heading.setStyleSheet("font-size: 12px; color: %s;" % theme.TEXT_SECONDARY)
            v.addWidget(heading)

            self._group = QButtonGroup(self)
            self._radio_single = QRadioButton("Single monitor")
            self._radio_mirror = QRadioButton("Mirror all")
            self._radio_pick = QRadioButton("Pick one")
            for r in (self._radio_single, self._radio_mirror, self._radio_pick):
                self._group.addButton(r)
                v.addWidget(r)
                # Help text per UI-SPEC Surface 1 (verbatim)
            self._help_single = QLabel("Show one server monitor at a time. Lowest bandwidth.")
            self._help_mirror = QLabel("Show every server monitor in one window. Matches client review workflows.")
            self._help_pick = QLabel("Choose a specific server monitor. Remembered across sessions.")
            for h in (self._help_single, self._help_mirror, self._help_pick):
                h.setStyleSheet("font-size: 12px; color: %s; padding-left: 24px;" % theme.TEXT_SECONDARY)
                v.addWidget(h)

            # Pick-one sub-selector — only visible when pick_one is chosen
            self._pick_label = QLabel("Which monitor?")
            self._pick_label.setStyleSheet("font-size: 12px; color: %s;" % theme.TEXT_SECONDARY)
            self._monitor_picker = MonitorSelector()
            self._monitor_picker.set_mode("radio")  # D-04
            self._pick_label.setVisible(False)
            self._monitor_picker.setVisible(False)
            v.addWidget(self._pick_label)
            v.addWidget(self._monitor_picker)

            for r in (self._radio_single, self._radio_mirror, self._radio_pick):
                r.toggled.connect(self._on_radio_toggled)
            self._monitor_picker.selection_changed.connect(self._on_pick_changed)

            # default state
            self._radio_mirror.setChecked(True)

        def set_mode(self, mode: str, picked_id: int = -1, picked_name: str = ""):
            """Pre-fill from bookmark."""
            mapping = {"single": self._radio_single, "mirror_all": self._radio_mirror, "pick_one": self._radio_pick}
            r = mapping.get(mode, self._radio_mirror)
            r.setChecked(True)
            if mode == "pick_one" and picked_id >= 0:
                # MonitorSelector.update_monitors must have been called by caller
                self._monitor_picker.set_picked(picked_id, picked_name)

        def update_monitors(self, monitors: list):
            self._monitor_picker.update_monitors(monitors)

        @property
        def monitor_mode(self) -> str:
            if self._radio_single.isChecked(): return "single"
            if self._radio_mirror.isChecked(): return "mirror_all"
            return "pick_one"

        @property
        def picked_monitor_id(self) -> int:
            sel = self._monitor_picker.selected_ids()
            return sel[0] if sel else -1

        @property
        def picked_monitor_name(self) -> str:
            sel = self._monitor_picker.selected_names()
            return sel[0] if sel else ""

        def _on_radio_toggled(self, checked: bool):
            if not checked:
                return  # only react to checked-side
            is_pick = self._radio_pick.isChecked()
            self._pick_label.setVisible(is_pick)
            self._monitor_picker.setVisible(is_pick)
            self.mode_changed.emit(self.monitor_mode, self.picked_monitor_id, self.picked_monitor_name)

        def _on_pick_changed(self, ids: list):
            if self._radio_pick.isChecked():
                self.mode_changed.emit("pick_one", self.picked_monitor_id, self.picked_monitor_name)

        def set_disabled_during_session(self, disabled: bool):
            """D-03 — gray out + ForbiddenCursor + tooltip mid-session."""
            self.setEnabled(not disabled)
            tip = "Disconnect and reconnect to change monitor mode." if disabled else ""
            for w in (self._radio_single, self._radio_mirror, self._radio_pick,
                      self._pick_label, self._monitor_picker):
                w.setToolTip(tip)
                if disabled:
                    w.setCursor(Qt.ForbiddenCursor)
                else:
                    w.unsetCursor()
            self.setAccessibleName("Monitor mode" + (", disabled during active session" if disabled else ""))
    ```
