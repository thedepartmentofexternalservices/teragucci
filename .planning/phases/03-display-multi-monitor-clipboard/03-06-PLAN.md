    **Step A — Create common/clipboard_chunks.py (NEW MODULE).** Per RESEARCH Code Example 3 lines 612-647:
    ```python
    """Phase 3 D-17 — clipboard chunk reassembler.

    Used by both client/protocol.py and server/session_runtime.py to
    reassemble base64 chunks from large clipboard payloads (esp. images).
    Indexed by chunk_index NOT arrival order so future QUIC multiplexed
    delivery doesn't corrupt assembly (Pitfall 6)."""
    from __future__ import annotations
    import time
    from dataclasses import dataclass, field
    from typing import Dict, Optional


    CHUNK_TIMEOUT_S = 30.0           # drop incomplete sequences after 30s
    MAX_TOTAL_CHUNKS = 256           # T-03-03 mitigation: cap memory exposure
    MAX_CHUNK_BYTES = 2 * 1024 * 1024  # 1MB target + slack for base64 expansion


    @dataclass
    class ClipboardChunkAssembler:
        """D-17 chunk-reassembly state.

        Indexed by chunk_index (NOT arrival order) so out-of-order
        delivery over future QUIC multiplexed streams works. Returns
        the full payload once all chunks present; None until then.
        """
        sequence_id: int
        total_chunks: int
        content_type: str
        _slots: Dict[int, str] = field(default_factory=dict)
        _started_at: float = field(default_factory=time.monotonic)

        def __post_init__(self):
            if self.total_chunks <= 0 or self.total_chunks > MAX_TOTAL_CHUNKS:
                raise ValueError(
                    f"total_chunks {self.total_chunks} out of range [1, {MAX_TOTAL_CHUNKS}]"
                )

        def add(self, chunk_index: int, data: str) -> Optional[str]:
            """Append chunk; return concatenated base64 string when complete."""
            if not (0 <= chunk_index < self.total_chunks):
                return None  # out-of-range; drop (Pitfall 6)
            if chunk_index in self._slots:
                return None  # duplicate; idempotent ignore
            if len(data) > MAX_CHUNK_BYTES:
                return None  # over per-chunk cap; drop
            self._slots[chunk_index] = data
            if len(self._slots) == self.total_chunks:
                return "".join(self._slots[i] for i in range(self.total_chunks))
            return None

        def is_stale(self, now: Optional[float] = None) -> bool:
            now = now if now is not None else time.monotonic()
            return (now - self._started_at) > CHUNK_TIMEOUT_S
    ```

    ```python
    PNG_MAGIC = b"\x89PNG\r\n\x1a\n"
    PNG_MAX_BYTES = 64 * 1024 * 1024  # D-14 — 64 MB cap

    def validate_png_payload(payload: bytes) -> bool:
        """D-16 defense-in-depth PNG validation.

        Returns True iff payload is a well-formed PNG within the size cap.
        Logs a structlog warning and returns False on any failure."""
        if len(payload) > PNG_MAX_BYTES:
            logger.warning("clipboard.image_oversize size=%d cap=%d",
                           len(payload), PNG_MAX_BYTES)
            return False
        if len(payload) < 8 or payload[:8] != PNG_MAGIC:
            logger.warning("clipboard.image_bad_magic prefix=%s",
                           payload[:8].hex() if payload else "empty")
            return False
        try:
            from PIL import Image
            import io
            Image.open(io.BytesIO(payload)).verify()
        except Exception as e:
            logger.warning("clipboard.image_pil_verify_failed error=%s", e)
            return False
        return True
    ```

       ```python
                   self._enqueue_chunked_clipboard(cs, content_type, data_b64)
               else:
                   msg_json = ClipboardMsg(
                       type=MsgType.CLIPBOARD_RECV, content_type="text/plain", data=payload
                   ).to_json()
                   asyncio.run_coroutine_threadsafe(cs.enqueue(msg_json), self._event_loop)

       def _enqueue_chunked_clipboard(self, cs, content_type: str, data_b64: str):
           """D-17 — split base64 payload into 1MB chunks; emit ClipboardChunkMsg per chunk."""
           CHUNK_BYTES = 1024 * 1024
           total = max(1, (len(data_b64) + CHUNK_BYTES - 1) // CHUNK_BYTES)
           seq = self._next_clipboard_seq()
           for i in range(total):
               chunk = data_b64[i * CHUNK_BYTES : (i + 1) * CHUNK_BYTES]
               msg_json = ClipboardChunkMsg(
                   sequence_id=seq, chunk_index=i, total_chunks=total,
                   content_type=content_type, data=chunk,
               ).to_json()
               asyncio.run_coroutine_threadsafe(cs.enqueue(msg_json), self._event_loop)

       def _next_clipboard_seq(self) -> int:
           if not hasattr(self, "_clipboard_seq"):
               self._clipboard_seq = 0
           self._clipboard_seq = (self._clipboard_seq + 1) & 0xFFFFFFFF
           return self._clipboard_seq
       ```
