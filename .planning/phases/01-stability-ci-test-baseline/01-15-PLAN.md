    ```python
    """Diagnostic bundle export (OBS-05).

    Produces a teraguchi-diag-<timestamp>.zip with:
      manifest.json + config/ + logs/ + state/ + system/ + protocol/

    Redaction rules per RESEARCH §"Diagnostic bundle (OBS-05) format":
      - TLS private keys → [REDACTED: TLS_PRIVATE_KEY]
      - Broker HMAC secret file → [REDACTED: BROKER_HMAC_SECRET]
      - users.json password hashes → [REDACTED: USER_PASSWORD_HASH]
      - Bookmarks password_encrypted → [REDACTED: BOOKMARK_PASSWORD]
      - Env vars matching *_PASSWORD|*_SECRET|*_KEY|*_TOKEN → value [REDACTED]

    Regression test `test_no_secret_strings_leak` fails if any literal secret
    appears in the zip contents.
    """
    from __future__ import annotations

    import datetime
    import io
    import json
    import logging
    import os
    import pathlib
    import platform
    import re
    import subprocess
    import sys
    import zipfile
    from typing import Any, Dict, Optional

    logger = logging.getLogger("teraguchi.diagnostic_bundle")

    # Env var name patterns for redaction
    _SECRET_ENV_RE = re.compile(r".*(_PASSWORD|_SECRET|_KEY|_TOKEN|_PIN).*", re.IGNORECASE)

    # JSON field names to redact (case-insensitive substring)
    _JSON_REDACT_KEYS = {"password", "password_hash", "password_encrypted",
                         "token", "secret", "private_key", "tls_key",
                         "credential", "pin"}


    def _redact_dict(data: Any) -> Any:
        """Recursively redact password/secret/token/key fields."""
        if isinstance(data, dict):
            out = {}
            for k, v in data.items():
                lowered = str(k).lower()
                if any(needle in lowered for needle in _JSON_REDACT_KEYS):
                    out[k] = "[REDACTED]"
                else:
                    out[k] = _redact_dict(v)
            return out
        if isinstance(data, list):
            return [_redact_dict(x) for x in data]
        return data


    def _redact_text(text: str, category: str = "") -> str:
        """Scan text for secret-looking lines and redact the value portion.

        Hook for file types where structured redaction isn't applicable
        (TOML configs, free-form logs)."""
        patterns = [
            (re.compile(r"^(.*(?:password|secret|token|private_key|tls_key|credential)\s*[=:]\s*).+$",
                        re.IGNORECASE | re.MULTILINE), r"\1[REDACTED]"),
        ]
        for pat, repl in patterns:
            text = pat.sub(repl, text)
        return text


    def _probe_subprocess(argv: list, timeout: float = 5.0) -> str:
        """Run a subprocess probe; return output or an error string. Never raises."""
        try:
            result = subprocess.run(argv, capture_output=True, text=True,
                                     timeout=timeout, check=False)
            return result.stdout + ("\n---STDERR---\n" + result.stderr if result.stderr else "")
        except FileNotFoundError:
            return f"[not-available: {argv[0]}]"
        except subprocess.TimeoutExpired:
            return f"[timeout: {argv}]"
        except Exception as e:
            return f"[probe error: {e}]"


    def _env_vars_redacted() -> str:
        lines = []
        for k, v in sorted(os.environ.items()):
            if not k.startswith("TERAGUCHI_"):
                continue
            if _SECRET_ENV_RE.match(k):
                lines.append(f"{k}=[REDACTED]")
            else:
                lines.append(f"{k}={v}")
        return "\n".join(lines) + "\n"


    def _manifest(tier: str) -> Dict[str, Any]:
        import socket
        return {
            "version": "3.0.0",
            "timestamp": datetime.datetime.now(datetime.timezone.utc).isoformat(),
            "tier": tier,
            "hostname": socket.gethostname(),
            "os": f"{platform.system()} {platform.release()}",
            "python_version": sys.version,
            "cwd": str(pathlib.Path.cwd()),
        }


    def build_bundle(
        out_path: str,
        *,
        tier: str = "server",
        config_files: Optional[list] = None,
        log_files: Optional[list] = None,
        live_state: Optional[Dict[str, Any]] = None,
    ) -> pathlib.Path:
        """Build the diagnostic zip at out_path. Returns the path.

        Args:
            out_path: where to write the zip. If empty, derives from ~ + timestamp.
            tier: "server" / "client" / "broker"
            config_files: list of (display_name, path_on_disk) for config/ section
            log_files: list of (display_name, path_on_disk) for logs/ section
            live_state: dict of live runtime state (FSM states, queue depths, etc.)
        """
        if not out_path:
            ts = datetime.datetime.now().strftime("%Y%m%dT%H%M%SZ")
            out_path = str(pathlib.Path.home() / f"teraguchi-diag-{ts}.zip")
        out = pathlib.Path(out_path)
        out.parent.mkdir(parents=True, exist_ok=True)

        with zipfile.ZipFile(out, "w", zipfile.ZIP_DEFLATED) as zf:
            zf.writestr("manifest.json", json.dumps(_manifest(tier), indent=2))

            # config/
            zf.writestr("config/env-vars-redacted.txt", _env_vars_redacted())
            if config_files:
                for name, src in config_files:
                    try:
                        text = pathlib.Path(src).read_text()
                        redacted = _redact_text(text, category=name)
                        zf.writestr(f"config/{name}", redacted)
                    except FileNotFoundError:
                        zf.writestr(f"config/{name}", f"[not-found: {src}]\n")
                    except Exception as e:
                        zf.writestr(f"config/{name}", f"[read error: {e}]\n")

            # logs/
            if log_files:
                for name, src in log_files:
                    try:
                        p = pathlib.Path(src)
                        data = p.read_bytes()[-10 * 1024 * 1024:]   # last 10 MB
                        zf.writestr(f"logs/{name}", data)
                    except FileNotFoundError:
                        zf.writestr(f"logs/{name}", f"[not-found: {src}]\n")

            # state/
            state = live_state or {}
            for section in ("fsm-states", "health-stats", "queue-depths",
                             "encoder-state"):
                content = state.get(section, {})
                zf.writestr(f"state/{section}.json",
                            json.dumps(_redact_dict(content), indent=2))

            # system/
            zf.writestr("system/python-packages.txt",
                        _probe_subprocess([sys.executable, "-m", "pip", "freeze"]))
            zf.writestr("system/ffmpeg-version.txt",
                        _probe_subprocess(["ffmpeg", "-version"]))
            if platform.system() == "Darwin":
                zf.writestr("system/gpu.txt",
                            _probe_subprocess(["system_profiler", "SPDisplaysDataType"]))
            else:
                zf.writestr("system/gpu.txt",
                            _probe_subprocess(["nvidia-smi"]))
            zf.writestr("system/uname.txt", _probe_subprocess(["uname", "-a"]))
            zf.writestr("system/uptime.txt", _probe_subprocess(["uptime"]))

            # protocol/
            last_hello = state.get("last_hello")
            if last_hello:
                zf.writestr("protocol/last-hello.json",
                            json.dumps(_redact_dict(last_hello), indent=2))

        logger.info("diag.bundle_created path=%s size=%d", str(out), out.stat().st_size)
        return out
    ```

    ```python
    """OBS-05 + T-1-05 — diagnostic bundle content + redaction."""
    import json
    import pathlib
    import zipfile

    import pytest

    from common.diagnostic_bundle import build_bundle, _redact_dict, _redact_text


    def test_build_bundle_writes_zip(tmp_path):
        out = build_bundle(str(tmp_path / "diag.zip"), tier="server")
        assert out.exists()
        assert out.stat().st_size > 0


    def test_bundle_contains_expected_sections(tmp_path):
        out = build_bundle(str(tmp_path / "diag.zip"), tier="server")
        with zipfile.ZipFile(out) as zf:
            names = set(zf.namelist())
        assert "manifest.json" in names
        assert "config/env-vars-redacted.txt" in names
        assert any(n.startswith("state/") for n in names)
        assert any(n.startswith("system/") for n in names)


    def test_redact_dict_handles_nested():
        data = {
            "user": "alice",
            "password": "hunter2",
            "config": {"api_key": "secret-key-xyz", "ok_field": "visible"},
            "tokens": [{"token": "t1"}, {"other": "ok"}],
        }
        redacted = _redact_dict(data)
        assert redacted["user"] == "alice"
        assert redacted["password"] == "[REDACTED]"
        assert redacted["config"]["api_key"] == "[REDACTED]"
        assert redacted["config"]["ok_field"] == "visible"
        assert redacted["tokens"][0]["token"] == "[REDACTED]"
        assert redacted["tokens"][1]["other"] == "ok"


    def test_redact_text_replaces_inline_values():
        text = "password=hunter2\napi_key = abc123\nlog_level=INFO\n"
        redacted = _redact_text(text)
        assert "hunter2" not in redacted
        assert "abc123" not in redacted
        assert "INFO" in redacted
        assert "[REDACTED]" in redacted


    def test_no_secret_strings_leak(tmp_path, monkeypatch):
        """T-1-05 regression — secrets set in env + config files must NOT
        appear as plaintext in the zip."""
        SECRETS = ["hunter2-very-secret", "broker-hmac-deadbeef", "tls-pk-abc123"]

        # 1. Plant in env
        monkeypatch.setenv("TERAGUCHI_BROKER_SECRET", SECRETS[1])
        monkeypatch.setenv("TERAGUCHI_TLS_KEY", SECRETS[2])

        # 2. Plant in a mock config file
        cfg = tmp_path / "server.toml"
        cfg.write_text(f"""
# Fake config
password = "{SECRETS[0]}"
tls_key = "{SECRETS[2]}"
listen_port = 443
""")

        # 3. Plant in live state dict
        live = {
            "fsm-states": {"client1": {"state": "streaming", "token": SECRETS[1]}},
            "last_hello": {"version": "3.0", "password_hash": SECRETS[0]},
        }

        # 4. Build bundle
        out = build_bundle(str(tmp_path / "diag.zip"), tier="server",
                           config_files=[("server.toml", str(cfg))],
                           live_state=live)

        # 5. Extract all members to text and grep for plaintext secrets
        with zipfile.ZipFile(out) as zf:
            for name in zf.namelist():
                try:
                    content = zf.read(name).decode(errors="ignore")
                except Exception:
                    continue
                for secret in SECRETS:
                    assert secret not in content, (
                        f"PLAINTEXT SECRET LEAKED in {name}: contains {secret!r}"
                    )


    def test_bundle_tier_in_manifest(tmp_path):
        out = build_bundle(str(tmp_path / "d.zip"), tier="client")
        with zipfile.ZipFile(out) as zf:
            m = json.loads(zf.read("manifest.json"))
        assert m["tier"] == "client"
    ```
