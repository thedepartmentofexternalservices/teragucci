    ```python
    """Shared structlog configuration (OBS-01).

    Imported by all three entry points (server, client, broker) and by the
    smoke harness. Idempotent — safe to call configure() multiple times.

    Canonical schema per CONTEXT.md §"Claude's Discretion" line 83:
      Required on every event:
        - event   (dotted short name, first positional arg to log.info(...))
        - phase   ("server" | "client" | "broker")  bound via contextvars.bind_contextvars in configure()
        - level   (debug/info/warning/...)          via structlog.processors.add_log_level
        - ts      (ISO 8601 UTC)                    via TimeStamper(fmt="iso", utc=True, key="ts")
        - logger  (dotted logger name)              via structlog.stdlib.add_logger_name

      Required on session-scoped events (bound inside a session context):
        - session_id     (16-hex)   bound per SessionRuntime
        - client_id      (string)   bound per ClientSession handler
        - stage          (capture/encode/...)       bound per StageTimer block

      Optional (bind per-context):
        username, state, prev_state, transport, latency_ms,
        frame_size, keyframe, err_class, err_code.

    Session-scope enforcement: when a caller tags an event with
    session_scope=True, a missing session_id / client_id / stage triggers
    a warn-level log ("log.session_scope_missing_fields") on the same emit.
    The original event is still delivered (never dropped); the warning is
    a diagnostic nudge for incomplete wiring.

    Redaction (T-1-04): any event dict key matching REDACT_KEYS (case-insensitive
    substring match) has its VALUE replaced with "[REDACTED]" BEFORE JSONRenderer.
    """
    from __future__ import annotations

    import contextvars
    import logging
    import sys
    from typing import Any

    import structlog
    from structlog.typing import EventDict, Processor, WrappedLogger

    # session_id contextvar (bound per-SessionRuntime in server; per-tab in client)
    session_id_var: contextvars.ContextVar[str] = contextvars.ContextVar(
        "session_id", default="")

    # Case-insensitive substring match — catches "password", "api_key", "TOKEN",
    # "user_credentials", "broker_secret", "private_key", "pin_code", etc.
    REDACT_KEYS: tuple[str, ...] = (
        "password", "credential", "token", "secret", "key", "pin",
    )

    # Fields required on every event tagged session_scope=True
    SESSION_SCOPE_REQUIRED: tuple[str, ...] = ("session_id", "client_id", "stage")


    def _redact_secrets(
        logger: WrappedLogger,
        method_name: str,
        event_dict: EventDict,
    ) -> EventDict:
        """Processor — strips any key containing a REDACT_KEYS substring.

        Runs BEFORE JSONRenderer per RESEARCH Pitfall 2. Replaces values with
        the literal string "[REDACTED]" (never drops the key — visibility that
        redaction happened is itself useful signal).
        """
        out: dict[str, Any] = {}
        for k, v in event_dict.items():
            lowered = str(k).lower()
            if any(needle in lowered for needle in REDACT_KEYS):
                out[k] = "[REDACTED]"
            else:
                out[k] = v
        return out


    def _warn_session_scope_missing(
        logger: WrappedLogger,
        method_name: str,
        event_dict: EventDict,
    ) -> EventDict:
        """If session_scope=True is set, warn about any missing required fields.

        Does NOT drop the original event; emits a separate warn log with the
        list of missing fields. Keeps wiring honest without breaking runtime.
        """
        if event_dict.get("session_scope") is True:
            # Check merged contextvars (already applied upstream by merge_contextvars)
            missing = [
                field for field in SESSION_SCOPE_REQUIRED
                if not event_dict.get(field)
            ]
            if missing:
                try:
                    # Use stdlib logging to emit a side-channel WARN so we don't
                    # recurse into structlog's own processor chain.
                    logging.getLogger("teraguchi.logging").warning(
                        "log.session_scope_missing_fields missing=%s event=%s",
                        ",".join(missing),
                        event_dict.get("event", "<unknown>"),
                    )
                except Exception:
                    pass
        return event_dict


    _configured = False


    def configure(phase: str, *, verbose: bool = False) -> None:
        """Configure structlog once with the canonical CONTEXT.md schema.

        Args:
            phase: 'server', 'client', or 'broker'. Bound into contextvars so
                every event carries it. Entry points call this in main(). Name
                `phase` per CONTEXT.md §"Claude's Discretion" line 83 (the
                canonical user-locked field name — this is the deployment tier,
                NOT the GSD workflow phase).
            verbose: DEBUG level if True, else INFO.
        """
        global _configured
        if _configured:
            return
        _configured = True

        level = logging.DEBUG if verbose else logging.INFO
        processors: list[Processor] = [
            structlog.contextvars.merge_contextvars,        # MUST be before JSONRenderer (Pitfall 2)
            structlog.stdlib.add_logger_name,               # adds `logger` field (canonical schema)
            structlog.processors.add_log_level,
            structlog.processors.TimeStamper(fmt="iso", utc=True, key="ts"),
            structlog.processors.StackInfoRenderer(),
            structlog.processors.format_exc_info,
            structlog.processors.UnicodeDecoder(),
            _warn_session_scope_missing,                    # diagnostic — must run after merge_contextvars
            _redact_secrets,                                # T-1-04 — before renderer
            structlog.processors.JSONRenderer(sort_keys=True),
        ]
        structlog.configure(
            processors=processors,
            wrapper_class=structlog.make_filtering_bound_logger(level),
            context_class=dict,
            logger_factory=structlog.PrintLoggerFactory(file=sys.stderr),
            cache_logger_on_first_use=True,
        )
        # Bridge stdlib logging.getLogger(...) → structlog so existing logger.info()
        # call sites (documented in CONVENTIONS.md §"Logging") flow through the same pipeline.
        logging.basicConfig(
            format="%(message)s",
            stream=sys.stderr,
            level=level,
        )
        # Tag every emit with the deployment phase (server/client/broker).
        structlog.contextvars.bind_contextvars(phase=phase)


    def get_logger(name: str = "") -> Any:
        """Return a structlog BoundLogger.

        Use `name="server.broadcaster"` so logs get `logger=teraguchi.server.broadcaster`.
        Call sites that still use stdlib logging get the same output via basicConfig bridge.
        """
        return structlog.get_logger(f"teraguchi.{name}" if name else "teraguchi")


    class StageTimer:
        """Context manager that binds `stage` + emits `latency_ms` for the block.

        Usage:
            with StageTimer("capture"):
                raw = capture.capture_raw_bgra()

        On exit, logs a stage.timing event with the elapsed ms.
        """
        def __init__(self, stage: str):
            self.stage = stage
            self._log = get_logger("stage")

        def __enter__(self):
            import time
            self._t0 = time.perf_counter()
            structlog.contextvars.bind_contextvars(stage=self.stage)
            return self

        def __exit__(self, exc_type, exc, tb):
            import time
            elapsed_ms = (time.perf_counter() - self._t0) * 1000.0
            self._log.info("stage.timing", stage=self.stage, latency_ms=round(elapsed_ms, 3))
            structlog.contextvars.unbind_contextvars("stage")
            return False   # don't suppress
    ```
