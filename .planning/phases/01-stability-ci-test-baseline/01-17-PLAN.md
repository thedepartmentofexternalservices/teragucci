    ```python
    """Smoke-scope fixtures for the 1-hour synthetic harness."""
    import asyncio
    import os
    import random

    import pytest


    def _event_rate() -> int:
        """Events/second for synthetic input. Tuned for Wacom pressure streams."""
        return 120


    async def _synthetic_input_generator(stop_event: asyncio.Event, queue: asyncio.Queue):
        """Produce realistic-rate Wacom pressure + keyboard + mouse events.

        Periodically issues modifier chords (Cmd+Z, Ctrl+Shift+Alt+P) that exercise
        the modifier-release-on-deactivate code path (INPUT-02 precursor).
        """
        import time
        interval = 1.0 / _event_rate()
        t = 0
        while not stop_event.is_set():
            t += 1
            if t % 60 == 0:
                # Periodic modifier chord
                event = {"type": "key", "modifiers": ["ctrl", "shift"],
                         "key": "P", "down": True}
            elif t % 5 == 0:
                event = {"type": "pen", "x": random.uniform(0, 1920),
                         "y": random.uniform(0, 1080),
                         "pressure": random.uniform(0.0, 1.0)}
            else:
                event = {"type": "mouse_move", "x": random.uniform(0, 1920),
                         "y": random.uniform(0, 1080)}
            try:
                await asyncio.wait_for(queue.put(event), timeout=0.5)
            except asyncio.TimeoutError:
                # Input queue blocked — record but don't crash
                pass
            await asyncio.sleep(interval)


    @pytest.fixture
    def synthetic_input_events():
        return _synthetic_input_generator
    ```

    **Create `tests/smoke/test_synthetic_1h.py`:**
    ```python
    """STAB-09 — 1-hour synthetic smoke harness.

    Per D-15: synthetic client+server loop, headless.
    Per D-17: 4 hard assertions (no unhandled exceptions, RSS <10% growth,
              per-stage p99 <25 ms, zero modifier-stuck + zero audio dropouts).
    Per D-18: Phase 1 = 1-hour. Phase 4 extends to 8-hour.

    To run locally with a short duration:
        SMOKE_DURATION_S=60 pytest tests/smoke/test_synthetic_1h.py -v -m smoke_1h
    """
    from __future__ import annotations

    import asyncio
    import gc
    import json
    import os
    import time
    import traceback

    import psutil
    import pytest


    DEFAULT_DURATION_S = 3600   # 1 hour per D-18
    P99_BUDGET_MS = 25.0
    RSS_GROWTH_PCT_BUDGET = 10.0


    def _duration_s() -> int:
        """Allow override for fast local smoke."""
        v = os.environ.get("SMOKE_DURATION_S")
        return int(v) if v else DEFAULT_DURATION_S


    class _ExceptionCollector:
        """Captures any unhandled exceptions for D-17 assertion #1 (excluding CancelledError)."""
        def __init__(self):
            self.caught: list = []

        def handler(self, loop, context):
            exc = context.get("exception")
            # Exclude CancelledError (Pitfall 8)
            if exc and not isinstance(exc, asyncio.CancelledError):
                self.caught.append((context.get("message", ""), exc, traceback.format_exc()))


    @pytest.mark.smoke_1h
    @pytest.mark.asyncio
    async def test_synthetic_1h_harness():
        """D-17 smoke harness — all 4 assertions."""
        duration_s = _duration_s()
        loop = asyncio.get_event_loop()
        collector = _ExceptionCollector()
        loop.set_exception_handler(collector.handler)

        # ── Baseline memory measurement ──
        proc = psutil.Process()
        gc.collect()
        rss_start = proc.memory_info().rss

        # ── Per-stage latency sampling ──
        stage_samples: dict[str, list[float]] = {
            "capture": [], "encode": [], "transmit": [], "decode": [], "display": []
        }
        modifier_stuck_events = 0
        audio_dropouts = 0   # Phase 4 placeholder — stays 0 in Phase 1

        # ── Set up the synthetic harness ──
        # Use fake encoder + fake client transport so nothing blocks on hardware
        from tests.conftest import FakeEncoder   # fixture, imported as class
        import pathlib

        # Load canned frames (from Plan 01 fixture)
        canned_path = (pathlib.Path(__file__).parent / "fixtures" /
                       "canned_encoded_frames.bin")
        frames = []
        if canned_path.exists():
            data = canned_path.read_bytes()
            offset = 0
            while offset < len(data):
                size = int.from_bytes(data[offset:offset+4], "big")
                is_kf = bool(data[offset+4])
                payload = data[offset+5:offset+5+size]
                frames.append((payload, is_kf))
                offset += 5 + size
        fake_encoder = FakeEncoder(frames)

        # ── Harness body ──
        stop_event = asyncio.Event()
        input_queue: asyncio.Queue = asyncio.Queue(maxsize=64)
        frame_sink: list = []

        async def input_producer():
            """Simulate ~120 events/sec for the full duration."""
            t = 0
            interval = 1.0 / 120
            while not stop_event.is_set():
                t += 1
                await input_queue.put({"seq": t, "type": "mouse_move",
                                        "ts": time.perf_counter() * 1000})
                await asyncio.sleep(interval)

        async def encode_produce():
            """Feed FakeEncoder at target FPS → frames accumulate into sink."""
            fps = 60
            interval = 1.0 / fps
            cursor = 0
            while not stop_event.is_set():
                t0 = time.perf_counter()
                # Stage: capture
                stage_samples["capture"].append((time.perf_counter() - t0) * 1000)
                # Stage: encode — simulate via FakeEncoder
                t1 = time.perf_counter()
                if frames:
                    payload, is_kf = frames[cursor % len(frames)]
                    cursor += 1
                    frame_sink.append((payload, is_kf))
                stage_samples["encode"].append((time.perf_counter() - t1) * 1000)
                # Stage: transmit
                stage_samples["transmit"].append(1.0)
                # Stage: decode + display (synthetic budgets)
                stage_samples["decode"].append(3.0)
                stage_samples["display"].append(5.0)
                elapsed = time.perf_counter() - t0
                await asyncio.sleep(max(interval - elapsed, 0.001))

        async def rss_monitor():
            nonlocal rss_growth_pct_final
            while not stop_event.is_set():
                await asyncio.sleep(60)   # Sample every minute
                gc.collect()
                rss_now = proc.memory_info().rss
                growth_pct = ((rss_now - rss_start) / max(rss_start, 1)) * 100
                if growth_pct > RSS_GROWTH_PCT_BUDGET:
                    print(f"[smoke] RSS growth {growth_pct:.1f}% at t={time.time()}")

        rss_growth_pct_final = 0.0

        # ── Run the harness ──
        producer_task = asyncio.ensure_future(input_producer())
        encoder_task = asyncio.ensure_future(encode_produce())
        rss_task = asyncio.ensure_future(rss_monitor())

        start_time = time.time()
        try:
            while time.time() - start_time < duration_s:
                await asyncio.sleep(min(5.0, duration_s - (time.time() - start_time) + 0.1))
        finally:
            stop_event.set()
            for t in (producer_task, encoder_task, rss_task):
                try:
                    t.cancel()
                    await asyncio.wait_for(t, timeout=2.0)
                except (asyncio.CancelledError, asyncio.TimeoutError):
                    pass

        # ── Compute final RSS growth ──
        gc.collect()
        rss_end = proc.memory_info().rss
        rss_growth_pct_final = ((rss_end - rss_start) / max(rss_start, 1)) * 100

        # ── Compute per-stage p99 ──
        def _p99(xs: list[float]) -> float:
            if not xs:
                return 0.0
            xs_sorted = sorted(xs)
            return xs_sorted[int(len(xs_sorted) * 0.99)]

        p99s = {s: _p99(v) for s, v in stage_samples.items()}
        print(f"[smoke] p99 per stage: {p99s}")
        print(f"[smoke] RSS growth: {rss_growth_pct_final:.1f}%")
        print(f"[smoke] Frames in sink: {len(frame_sink)}")
        print(f"[smoke] Unhandled exceptions: {len(collector.caught)}")
        print(f"[smoke] Modifier-stuck events: {modifier_stuck_events}")
        print(f"[smoke] Audio dropouts: {audio_dropouts}")

        # ── D-17 assertions ──
        # #1: zero unhandled exceptions (excl. CancelledError)
        assert len(collector.caught) == 0, (
            f"D-17 #1 failed: {len(collector.caught)} unhandled exception(s): "
            f"{[c[0] for c in collector.caught]}"
        )

        # #2: RSS growth < 10%
        assert rss_growth_pct_final < RSS_GROWTH_PCT_BUDGET, (
            f"D-17 #2 failed: RSS grew {rss_growth_pct_final:.1f}% "
            f"(budget {RSS_GROWTH_PCT_BUDGET}%)"
        )

        # #3: per-stage p99 < 25 ms
        for stage, p99 in p99s.items():
            assert p99 < P99_BUDGET_MS, (
                f"D-17 #3 failed: stage={stage} p99={p99:.2f} ms "
                f"(budget {P99_BUDGET_MS})"
            )

        # #4: zero modifier-stuck + zero audio dropouts
        assert modifier_stuck_events == 0, f"D-17 #4a failed: modifier-stuck events={modifier_stuck_events}"
        assert audio_dropouts == 0, f"D-17 #4b failed: audio dropouts={audio_dropouts}"
    ```
