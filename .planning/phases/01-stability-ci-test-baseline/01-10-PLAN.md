    ```python
    """Stream loops (H264 + JPEG) — extracted from server/main.py::SessionRuntime.

    D-11 extraction. Preserves the existing behavior exactly:
      - `while self._running:` loop
      - FPS-pacing via `await asyncio.sleep(max(interval - elapsed, 0.001))`
      - Per-frame try/except around capture + encode (log and continue)

    Instrumentation (OBS-02 per-stage timing) lands in Plan 14; this module
    only does the extraction, no timing changes.
    """
    from __future__ import annotations

    import asyncio
    import logging
    import time
    from typing import Callable, Optional

    logger = logging.getLogger("teraguchi.server.stream_loop")


    class StreamLoop:
        """Capture → encode → dispatch loop. One instance per SessionRuntime."""

        def __init__(self, runtime: "SessionRuntime"):   # forward ref OK
            self._runtime = runtime
            self._running = False
            self._task: Optional[asyncio.Task] = None

        async def run_h264(self, fps: int) -> None:
            interval = 1.0 / fps
            while self._running:
                start = time.time()
                if self._runtime.clients and self._runtime.encoder:
                    try:
                        t0 = time.time()
                        raw = self._runtime.capture.capture_raw_bgra()
                        self._runtime.health.record_capture_time(
                            (time.time() - t0) * 1000)
                        self._runtime.encoder.feed_frame(raw)
                    except Exception as e:
                        logger.error("[%s] H264 capture error: %s",
                                     self._runtime.username, e)
                elapsed = time.time() - start
                await asyncio.sleep(max(interval - elapsed, 0.001))

        async def run_jpeg(self, fps: int) -> None:
            # Same shape as run_h264 — extract the JPEG variant body verbatim from
            # the original server/main.py. Preserve the dirty-rect / poll logic.
            # DO NOT simplify; CONCERNS.md flags the JPEG path as dead-weight but
            # Phase 1 faithful extraction keeps it.
            interval = 1.0 / fps
            while self._running:
                start = time.time()
                # ... paste existing _stream_jpeg body from server/main.py ...
                # ... (read the source; copy as-is, replacing `self` with `self._runtime` for shared state) ...
                await asyncio.sleep(max(interval - (time.time() - start), 0.001))

        def start(self, fps: int) -> None:
            self._running = True
            codec = getattr(self._runtime.quality, "codec", "h264").lower()
            if codec == "jpeg":
                self._task = asyncio.ensure_future(self.run_jpeg(fps))
            else:
                self._task = asyncio.ensure_future(self.run_h264(fps))

        def stop(self) -> None:
            self._running = False
            if self._task and not self._task.done():
                self._task.cancel()
    ```

    ```python
    """Monitor hot-plug detection — extracted from SessionRuntime."""
    from __future__ import annotations

    import asyncio
    import logging
    from dataclasses import asdict
    from typing import Optional

    from common.messages import MonitorListMsg

    logger = logging.getLogger("teraguchi.server.monitor_hotplug")


    class MonitorHotplug:
        def __init__(self, runtime: "SessionRuntime"):
            self._runtime = runtime
            self._running = False
            self._task: Optional[asyncio.Task] = None

        async def run(self) -> None:
            while self._running:
                await asyncio.sleep(5.0)
                if self._runtime.capture and self._runtime.capture.detect_hotplug():
                    monitors = [asdict(m) for m in self._runtime.capture.list_monitors()]
                    msg_json = MonitorListMsg(monitors=monitors).to_json()
                    for ws, cs in list(self._runtime.clients.items()):
                        if cs.authenticated:
                            try:
                                await cs.enqueue(msg_json)
                            except Exception:
                                pass
                    if self._runtime.encoder_lifecycle:
                        self._runtime.encoder_lifecycle.restart()

        def start(self) -> None:
            self._running = True
            self._task = asyncio.ensure_future(self.run())

        def stop(self) -> None:
            self._running = False
            if self._task and not self._task.done():
                self._task.cancel()
    ```

    ```python
    from __future__ import annotations

    import logging
    from typing import Optional

    logger = logging.getLogger("teraguchi.server.encoder_lifecycle")


    class EncoderLifecycle:
        """Owns the VideoEncoder (or JpegFallbackEncoder) singleton for a SessionRuntime."""

        def __init__(self, runtime: "SessionRuntime"):
            self._runtime = runtime
            self.encoder = None

        def spawn(self) -> None:
            """Create the encoder matching the quality codec + caps."""
            # Paste the existing encoder-selection logic from
            # server/main.py::SessionRuntime.__init__ (~:146-162 in the original).
            # Preserve the `codec in (...) and ffmpeg_caps.get(codec, False)` check.
            # Call self.encoder.start(self._runtime._on_encoded_frame).
            # Set self._runtime.health.current_resolution.
            from server.video_encoder import VideoEncoder
            self.encoder = VideoEncoder(
                self._runtime.capture.width,
                self._runtime.capture.height,
                self._runtime.quality,
                available_encoders=None,
            )
            self.encoder.start(self._runtime._on_encoded_frame)
            self._runtime.health.current_resolution = (
                f"{self._runtime.capture.width}x{self._runtime.capture.height}"
            )

        def restart(self) -> None:
            """Stop current, spawn new with same quality settings. Used by
            quality-slider changes and monitor hot-plug."""
            old_available = None
            if self.encoder:
                old_available = self.encoder._available
                self.encoder.stop()
            from server.video_encoder import VideoEncoder
            self.encoder = VideoEncoder(
                self._runtime.capture.width,
                self._runtime.capture.height,
                self._runtime.quality,
                available_encoders=old_available,
            )
            self.encoder.start(self._runtime._on_encoded_frame)
            self._runtime.health.current_resolution = (
                f"{self._runtime.capture.width}x{self._runtime.capture.height}"
            )

        def request_keyframe(self) -> None:
            """Forwarder — for STAB-04 IDR-on-drop wiring."""
            if self.encoder:
                self.encoder.request_keyframe()

        def stop(self) -> None:
            if self.encoder:
                try:
                    self.encoder.stop()
                except Exception as e:
                    logger.warning("encoder stop error: %s", e)
                self.encoder = None
    ```
