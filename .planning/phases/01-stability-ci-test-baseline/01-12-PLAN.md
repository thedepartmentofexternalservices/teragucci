    import logging
    from typing import Optional

    from PySide6.QtCore import QObject, Signal, Qt
    from PySide6.QtWidgets import QPushButton, QTabWidget, QWidget

    logger = logging.getLogger("teraguchi.client.tab_manager")


    class TabManager(QObject):
        """Owns the central QTabWidget and tab lifecycle.

        Emits Qt Signals for tab_added / tab_closed / tab_switched so MainWindow
        reacts without knowing about the widget internals.
        """
        tab_added = Signal(int)       # tab index
        tab_closed = Signal(int)
        tab_switched = Signal(int)

        def __init__(self, parent: QObject):
            super().__init__(parent)
            self.widget = QTabWidget()
            self.widget.setTabsClosable(True)
            self.widget.setMovable(True)
            self.widget.setDocumentMode(True)
            self.widget.tabCloseRequested.connect(self._close_tab)
            self.widget.currentChanged.connect(self._on_tab_changed)
            self.widget.setCornerWidget(self._make_new_tab_btn(), Qt.TopRightCorner)

            # Paste existing _make_new_tab_btn from client/main.py

    """ConnectionSupervisor — STAB-08 client-side reconnect policy + ClientFSM driver.

    Owns exponential backoff (base 1s, ×2, cap 30s), ±25% jitter, retry cap (20).
    Drives ClientFSM state transitions on connect/disconnect/retry events.

    Extracts reconnect logic currently sprinkled across client/protocol.py::_run_loop.
    Testable in isolation with a fake transport_factory.
    """
    from __future__ import annotations

    import asyncio
    import logging
    import random
    from typing import Awaitable, Callable, Optional

    from common.session_fsm import ClientFSM

    logger = logging.getLogger("teraguchi.client.connection_supervisor")


    TransportFactory = Callable[[], Awaitable[None]]


    class ConnectionSupervisor:
        """Async supervisor — call .connect() and it stays connected until
        .close() is called or max_retries is exceeded."""

        def __init__(
            self,
            transport_factory: TransportFactory,
            *,
            fsm: Optional[ClientFSM] = None,
            max_retries: int = 20,
            base_delay: float = 1.0,
            max_delay: float = 30.0,
            jitter_pct: float = 0.25,
        ):
            self._factory = transport_factory
            self.fsm = fsm if fsm is not None else ClientFSM()
            self.max_retries = max_retries
            self.base_delay = base_delay
            self.max_delay = max_delay
            self.jitter_pct = jitter_pct

            self._closing = False
            self._retry_count = 0
            self._current_delay = base_delay

        @property
        def is_reconnecting(self) -> bool:
            return self.fsm.current_state.id == "reconnecting"

        @property
        def retry_count(self) -> int:
            return self._retry_count

        def _next_delay(self) -> float:
            """Exponential backoff + random ±jitter_pct."""
            delay = min(self._current_delay, self.max_delay)
            spread = delay * self.jitter_pct
            jittered = delay + random.uniform(-spread, spread)
            self._current_delay = min(self._current_delay * 2, self.max_delay)
            return max(0.0, jittered)

        def _reset_backoff(self):
            self._current_delay = self.base_delay
            self._retry_count = 0

        async def connect(self):
            """Run the factory; on failure, retry with backoff until max_retries."""
            while not self._closing:
                try:
                    try:
                        self.fsm.send("connect_requested")
                    except Exception:
                        pass
                    logger.info("supervisor.connect_attempt retry=%d", self._retry_count)
                    await self._factory()
                    # Factory returned cleanly — this means the transport layer
                    # exited normally. Reset backoff for the next cycle.
                    self._reset_backoff()
                    return
                except Exception as e:
                    logger.warning("supervisor.transport_error err=%s", e)
                    try:
                        self.fsm.send("transport_lost")
                    except Exception:
                        pass
                    self._retry_count += 1
                    if self._retry_count > self.max_retries:
                        logger.error("supervisor.max_retries attempts=%d", self._retry_count)
                        try:
                            self.fsm.send("max_retries")
                        except Exception:
                            pass
                        return
                    delay = self._next_delay()
                    logger.info("supervisor.backoff delay_s=%.2f retry=%d",
                                delay, self._retry_count)
                    await asyncio.sleep(delay)

        async def close(self):
            self._closing = True
            try:
                self.fsm.send("user_quit")
            except Exception:
                pass
    ```

    """STAB-08 — ConnectionSupervisor backoff, jitter, retry cap, FSM drive."""
    import asyncio

    import pytest

    from client.connection_supervisor import ConnectionSupervisor
    from common.session_fsm import ClientFSM


    async def _boom_transport():
        raise RuntimeError("transport error")


    async def _ok_transport():
        return None


    @pytest.mark.asyncio
    async def test_retry_cap_triggers_max_retries():
        fsm = ClientFSM()
        sup = ConnectionSupervisor(_boom_transport, fsm=fsm,
                                     max_retries=3, base_delay=0.001,
                                     max_delay=0.001, jitter_pct=0.0)
        await sup.connect()
        assert sup.retry_count > 3
        assert fsm.current_state.id == "closed"


    @pytest.mark.asyncio
    async def test_successful_connect_resets_backoff():
        sup = ConnectionSupervisor(_ok_transport, base_delay=1.0)
        await sup.connect()
        assert sup.retry_count == 0


    @pytest.mark.asyncio
    async def test_exponential_backoff_sequence():
        """Without jitter, backoff is deterministic: base, 2*base, 4*base, ... capped at max."""
        sup = ConnectionSupervisor(_boom_transport, max_retries=100,
                                     base_delay=1.0, max_delay=8.0, jitter_pct=0.0)
        delays = []
        for _ in range(5):
            delays.append(sup._next_delay())
        # Without jitter: 1, 2, 4, 8, 8 (capped)
        assert delays == [1.0, 2.0, 4.0, 8.0, 8.0]


    @pytest.mark.asyncio
    async def test_jitter_produces_variance():
        """With jitter_pct=0.25 the same delay produces different values."""
        sup = ConnectionSupervisor(_boom_transport, base_delay=1.0,
                                     max_delay=1.0, jitter_pct=0.25)
        observations = set()
        for _ in range(20):
            sup._current_delay = 1.0   # reset
            observations.add(round(sup._next_delay(), 3))
        # Observed at least 3 distinct values (randomness should produce ≥5 usually)
        assert len(observations) >= 3


    @pytest.mark.asyncio
    async def test_drives_client_fsm_on_transport_error():
        fsm = ClientFSM()
        sup = ConnectionSupervisor(_boom_transport, fsm=fsm,
                                     max_retries=1, base_delay=0.001,
                                     max_delay=0.001, jitter_pct=0.0)
        # Drive into a state where transport_lost is a valid transition
        fsm.send("connect_requested")
        fsm.send("tls_ok")
        fsm.send("auth_ok")
        fsm.send("hello_received")
        assert fsm.current_state.id == "streaming"

        await sup.connect()
        # After transport errors, FSM should have progressed through reconnecting → closed
        assert fsm.current_state.id in ("reconnecting", "closed")


    @pytest.mark.asyncio
    async def test_is_reconnecting_property():
        fsm = ClientFSM()
        sup = ConnectionSupervisor(_ok_transport, fsm=fsm)
        # Initially disconnected
        assert sup.is_reconnecting is False
        # Transition into reconnecting
        fsm.send("connect_requested")
        fsm.send("transport_lost")
        assert fsm.current_state.id == "reconnecting"
        assert sup.is_reconnecting is True


    @pytest.mark.asyncio
    async def test_close_transitions_to_closed():
        fsm = ClientFSM()
        sup = ConnectionSupervisor(_ok_transport, fsm=fsm)
        await sup.close()
        assert sup._closing is True
        assert fsm.current_state.id == "closed"

    """STAB-08 integration — supervisor-driven reconnect against a loopback server."""
    import asyncio
    import ssl

    import pytest
    import websockets

    from client.connection_supervisor import ConnectionSupervisor
    from common.session_fsm import ClientFSM


    def _server_ctx(cert, key):
        ctx = ssl.SSLContext(ssl.PROTOCOL_TLS_SERVER)
        ctx.load_cert_chain(str(cert), str(key))
        ctx.minimum_version = ssl.TLSVersion.TLSv1_2
        return ctx


    @pytest.mark.asyncio
    async def test_supervisor_reconnects_after_transport_lost(
        tls_ca_and_cert, free_port
    ):
        """Server drops the connection once, supervisor reconnects, second try succeeds."""
        accept_count = 0

        async def server_handler(ws):
            nonlocal accept_count
            accept_count += 1
            if accept_count == 1:
                # First accept: close immediately (simulate flap)
                await ws.close()
            else:
                # Second accept: stay open for 200 ms then close cleanly
                await asyncio.sleep(0.2)

        ssl_srv = _server_ctx(tls_ca_and_cert["server_cert"], tls_ca_and_cert["server_key"])

        async with websockets.serve(server_handler, "127.0.0.1", free_port, ssl=ssl_srv):
            cli_ctx = ssl.create_default_context(cafile=str(tls_ca_and_cert["ca_cert"]))
            connects_seen = 0

            async def transport_factory():
                nonlocal connects_seen
                connects_seen += 1
                async with websockets.connect(
                    f"wss://127.0.0.1:{free_port}", ssl=cli_ctx
                ) as ws:
                    # Drain messages until the server closes
                    async for _ in ws:
                        pass

            fsm = ClientFSM()
            sup = ConnectionSupervisor(
                transport_factory, fsm=fsm,
                max_retries=3, base_delay=0.01, max_delay=0.05, jitter_pct=0.0,
            )
            await sup.connect()

        # Supervisor should have attempted ≥ 2 connects (original + at least 1 retry)
        assert connects_seen >= 2
        assert accept_count >= 2


    @pytest.mark.asyncio
    async def test_supervisor_gives_up_after_max_retries(tls_ca_and_cert, free_port):
        """Port with nothing listening → retries exhaust → FSM closed."""
        # Do NOT start a server. Connection will fail every time.
        cli_ctx = ssl.create_default_context(cafile=str(tls_ca_and_cert["ca_cert"]))

        async def transport_factory():
            async with websockets.connect(
                f"wss://127.0.0.1:{free_port}", ssl=cli_ctx
            ) as ws:
                async for _ in ws:
                    pass

        fsm = ClientFSM()
        sup = ConnectionSupervisor(
            transport_factory, fsm=fsm,
            max_retries=2, base_delay=0.001, max_delay=0.001, jitter_pct=0.0,
        )
        await sup.connect()

        assert sup.retry_count > 2
        assert fsm.current_state.id == "closed"
    ```
